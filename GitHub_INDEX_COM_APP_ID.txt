  <script>
    const APP_URL = 'https://script.google.com/a/macros/appolus.com.br/s/AKfycbzzTf9IIJ2VRR4g3qiKezsuJvp20AnN8pjXRvvOCrJxGllzTPQohTYXkdAFExbEQ2Q_NA/exec';
    const frame = document.getElementById('appFrame');
    const loading = document.getElementById('loading');
    const offline = document.getElementById('offline');
    const refreshBtn = document.getElementById('refreshBtn');
    const openBtn = document.getElementById('openBtn');
    const installBox = document.getElementById('installBox');
    const installBtn = document.getElementById('installBtn');
    const installBtnTop = document.getElementById('installBtnTop');
    const dismissInstall = document.getElementById('dismissInstall');

    let deferredPrompt = null;

    function atualizarConexao() {
      offline.classList.toggle('show', !navigator.onLine);
    }

    function recarregarSistema() {
      loading.classList.remove('hide');
      const separador = APP_URL.includes('?') ? '&' : '?';
      frame.src = APP_URL + separador + '_pwa=' + Date.now();
    }

    frame.addEventListener('load', () => {
      setTimeout(() => loading.classList.add('hide'), 250);
    });

    refreshBtn.addEventListener('click', recarregarSistema);

    openBtn.addEventListener('click', () => {
      window.open(APP_URL, '_blank', 'noopener,noreferrer');
    });

    window.addEventListener('online', () => {
      atualizarConexao();
      recarregarSistema();
    });

    window.addEventListener('offline', atualizarConexao);

    window.addEventListener('beforeinstallprompt', (event) => {
      event.preventDefault();
      deferredPrompt = event;
      installBox.classList.add('show');
      installBtnTop.hidden = false;
    });

    async function instalar() {
      if (!deferredPrompt) {
        return;
      }

      deferredPrompt.prompt();
      await deferredPrompt.userChoice;
      deferredPrompt = null;
      installBox.classList.remove('show');
      installBtnTop.hidden = true;
    }

    installBtn.addEventListener('click', instalar);
    installBtnTop.addEventListener('click', instalar);
    dismissInstall.addEventListener('click', () => installBox.classList.remove('show'));

    window.addEventListener('appinstalled', () => {
      installBox.classList.remove('show');
      installBtnTop.hidden = true;
    });

    if ('serviceWorker' in navigator) {
      window.addEventListener('load', async () => {
        try {
          const registration = await navigator.serviceWorker.register('./service-worker.js');
          registration.update();
        } catch (erro) {
          console.error('Falha ao registrar service worker:', erro);
        }
      });
    }


    /* ============================================================
       ONESIGNAL / WEB PUSH
       ============================================================
       Substitua APENAS o valor abaixo pelo App ID do OneSignal.
       O App ID é público; a REST API Key fica somente no Code.gs
       através das Propriedades do script.
       ============================================================ */
    const ONESIGNAL_APP_ID = '0b01ec0c-8658-4593-94a8-9b282c52ca43';

    let oneSignalPronto = false;
    let usuarioPwaAtual = null;

    const notifyBtn =
      document.getElementById('notifyBtn');

    function oneSignalConfigurado() {
      return (
        ONESIGNAL_APP_ID &&
        ONESIGNAL_APP_ID !==
          '0b01ec0c-8658-4593-94a8-9b282c52ca43'
      );
    }

    if (oneSignalConfigurado()) {
      window.OneSignalDeferred =
        window.OneSignalDeferred || [];

      window.OneSignalDeferred.push(
        async function(OneSignal) {
          try {
            await OneSignal.init({
              appId: ONESIGNAL_APP_ID,
              serviceWorkerPath:
                'App-Atestado/push/onesignal/OneSignalSDKWorker.js',
              serviceWorkerParam: {
                scope:
                  '/App-Atestado/push/onesignal/'
              },
              notifyButton: {
                enable: false
              }
            });

            oneSignalPronto = true;

            if (
              usuarioPwaAtual &&
              usuarioPwaAtual.externalId
            ) {
              await OneSignal.login(
                usuarioPwaAtual.externalId
              );
            }

            atualizarBotaoNotificacao();
          } catch (erro) {
            console.error(
              'Falha ao inicializar notificações:',
              erro
            );
          }
        }
      );
    } else if (notifyBtn) {
      notifyBtn.title =
        'Configure o App ID do OneSignal primeiro';
    }

    window.addEventListener(
      'message',
      async function(event) {
        const dados =
          event && event.data
            ? event.data
            : {};

        if (
          dados.tipo ===
          'APPOLUS_PWA_USUARIO'
        ) {
          usuarioPwaAtual = {
            nome:
              String(dados.nome || ''),
            externalId:
              String(dados.externalId || '')
          };

          if (oneSignalPronto) {
            window.OneSignalDeferred.push(
              async function(OneSignal) {
                try {
                  await OneSignal.login(
                    usuarioPwaAtual.externalId
                  );

                  atualizarBotaoNotificacao();
                } catch (erro) {
                  console.error(
                    'Falha ao identificar usuário nas notificações:',
                    erro
                  );
                }
              }
            );
          }
        }

        if (
          dados.tipo ===
          'APPOLUS_PWA_LOGOUT'
        ) {
          usuarioPwaAtual = null;

          if (oneSignalPronto) {
            window.OneSignalDeferred.push(
              async function(OneSignal) {
                try {
                  await OneSignal.logout();
                  atualizarBotaoNotificacao();
                } catch (erro) {
                  console.error(
                    'Falha ao encerrar usuário das notificações:',
                    erro
                  );
                }
              }
            );
          }
        }
      }
    );

    async function ativarNotificacoesPWA() {
      if (!oneSignalConfigurado()) {
        alert(
          'As notificações ainda precisam ser configuradas no PWA.'
        );
        return;
      }

      if (
        !usuarioPwaAtual ||
        !usuarioPwaAtual.externalId
      ) {
        alert(
          'Entre no Controle de Atendimentos antes de ativar as notificações.'
        );
        return;
      }

      window.OneSignalDeferred.push(
        async function(OneSignal) {
          try {
            await OneSignal.login(
              usuarioPwaAtual.externalId
            );

            await OneSignal.Notifications
              .requestPermission();

            atualizarBotaoNotificacao();
          } catch (erro) {
            console.error(erro);

            alert(
              'Não foi possível ativar as notificações neste aparelho.'
            );
          }
        }
      );
    }

    function atualizarBotaoNotificacao() {
      if (!notifyBtn) {
        return;
      }

      if (!oneSignalConfigurado()) {
        notifyBtn.textContent =
          '🔔 Notificações';
        return;
      }

      window.OneSignalDeferred.push(
        async function(OneSignal) {
          try {
            const permissao =
              OneSignal.Notifications.permission;

            const optIn =
              OneSignal.User.PushSubscription.optedIn;

            if (permissao && optIn) {
              notifyBtn.textContent =
                '🔔 Ativadas';
              notifyBtn.title =
                'Notificações ativas neste aparelho';
            } else {
              notifyBtn.textContent =
                '🔔 Notificações';
              notifyBtn.title =
                'Ativar notificações';
            }
          } catch (erro) {
            // Mantém o botão normal.
          }
        }
      );
    }

    if (notifyBtn) {
      notifyBtn.addEventListener(
        'click',
        ativarNotificacoesPWA
      );
    }

    atualizarConexao();

  </script>
